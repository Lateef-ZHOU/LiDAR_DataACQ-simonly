cmake_minimum_required(VERSION 3.28)
project(LiDAR_DataACQ_simonly VERSION 1.2.1 LANGUAGES CXX)
set(LIDAR_VERSION_STRING "${PROJECT_VERSION}-simonly")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use a separate build directory")
endif()
find_package(Qt6 6.11.2 REQUIRED COMPONENTS Gui Qml Quick GuiPrivate ShaderTools)
include(cmake/Dependencies.cmake)
file(GLOB_RECURSE headers CONFIGURE_DEPENDS "include/*.h")
file(GLOB_RECURSE views CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "include/*.qml")
foreach(view IN LISTS views)
    get_filename_component(name "${view}" NAME)
    set_source_files_properties("${view}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
qt_add_executable(LiDAR_DataACQ MANUAL_FINALIZATION ${headers} src/main.cpp src/ImportModules.cpp)
if(WIN32)
    configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
    target_sources(LiDAR_DataACQ PRIVATE src/LiDAR_DataACQ.rc "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
    set(LIDAR_FONT_FAMILY "Microsoft YaHei UI")
    set(LIDAR_CONTROLS_STYLE "FluentWinUI3")
else()
    set(LIDAR_FONT_FAMILY "Noto Sans CJK SC")
    set(LIDAR_CONTROLS_STYLE "Fusion")
endif()
qt_add_qml_module(LiDAR_DataACQ URI Lidar.SimOnly VERSION 1.0 RESOURCE_PREFIX /qt/qml
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Lidar/SimOnly"
    DEPENDENCIES ${sim_uris} IMPORT_PATH ${sim_import_paths} QML_FILES ${views})
configure_file(src/qtquickcontrols2.conf.in "${CMAKE_CURRENT_BINARY_DIR}/qtquickcontrols2.conf" @ONLY)
qt_add_resources(LiDAR_DataACQ controls_conf PREFIX / BASE "${CMAKE_CURRENT_BINARY_DIR}"
    FILES "${CMAKE_CURRENT_BINARY_DIR}/qtquickcontrols2.conf")
set_source_files_properties(config/simulation.example.json PROPERTIES QT_RESOURCE_ALIAS simulation.example.json)
qt_add_resources(LiDAR_DataACQ simulation_defaults PREFIX / FILES config/simulation.example.json)
target_include_directories(LiDAR_DataACQ PRIVATE include include/Simulation)
target_link_libraries(LiDAR_DataACQ PRIVATE ${sim_modules} ${sim_plugins})
target_compile_definitions(LiDAR_DataACQ PRIVATE QT_NO_KEYWORDS
    LIDAR_VERSION_STRING="${LIDAR_VERSION_STRING}" LIDAR_FONT_FAMILY="${LIDAR_FONT_FAMILY}")
if(MSVC)
    target_compile_definitions(LiDAR_DataACQ PRIVATE NOMINMAX)
    target_compile_options(LiDAR_DataACQ PRIVATE /W4)
    get_target_property(sources LiDAR_DataACQ SOURCES)
    foreach(source IN LISTS sources)
        if(source MATCHES "/[.]rcc/qmlcache/")
            set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS /wd4702)
        endif()
    endforeach()
else()
    target_compile_options(LiDAR_DataACQ PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_finalize_executable(LiDAR_DataACQ)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(WIN32)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/run_LiDAR_DataACQ.cmd"
        CONTENT "@echo off\r\nset \"PATH=$<TARGET_FILE_DIR:Qt6::Core>;%PATH%\"\r\n\"$<TARGET_FILE:LiDAR_DataACQ>\" %*\r\n"
        CONDITION $<CONFIG:Release>)
endif()
